# nu_e-e elastic scattering: tools against the tree-level reference

line --no_error --datafile B4_Sigma -x Energy -y SigmaPerE --hue Label --select Channel -s es_numu --select Variant -s default --plot_type line --logx --rangex 0.1 20 --lower ratio --reference Label --reference_value 'snufish-reference 0.1.0' --lower_labely 'Ratio to tree level' --labelx 'Neutrino energy (MeV)' --labely '$\sigma/E$ (cm$^2$/MeV)' --title '$\nu_\mu e^-$ elastic scattering' --output_suffix es_numu
# nu_e CC on 40Ar: nuclear models over their full tables; the ratio is taken where every model has a value (>= 5 MeV, MARLEY tables)
line --no_error --datafile B4_Sigma -x Energy -y Sigma --hue Model --select Channel -s cc_nue_ar40 --plot_type line --logy --rangex 1.5 20 --rangey 1e-45 1e-40 --lower ratio --reference Model --reference_value 'snowglobes: default' --lower_rangey 0 1.95 --lower_labely 'Ratio to default' --labelx 'Neutrino energy (MeV)' --labely '$\sigma$ (cm$^2$ per $^{40}$Ar)' --title '$\nu_e$ CC on $^{40}$Ar: nuclear models' --output_suffix ar40_models
# ES interaction spectrum in argon by component (reference cross sections, daytime)
line --no_error --datafile B5_Spectrum -x Energy -y Rate --hue Component --select SolarModel -s b16_gs98 --select Binning -s log --select Model -s 'snufish-reference: default' --select Detector -s argon --select Channel -s es --remove_value Component=all --plot_type step --logx --logy --rangex 0.05 20 --rangey 1e-3 1e7 --labelx 'Neutrino energy (MeV)' --labely 'Interactions / (kt yr MeV)' --title 'ES interactions in argon by flux component' --output_suffix es_argon_components
